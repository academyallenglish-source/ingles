sw.js 
'use strict';
const CACHE='active-lingua-3276bf36422e-'+self.registration.scope;
const PREFIX='active-lingua-';
const FILES=["./LEEME.txt", "./LICENCIAS.txt", "./assets/app.css", "./assets/app.js", "./assets/brand-0.png", "./assets/brand-1.png", "./assets/font-0ee86126fb55.ttf", "./assets/font-160517eddbca.ttf", "./assets/font-43adbc7c773e.ttf", "./assets/font-722d54da3dbf.ttf", "./assets/font-9c0e6cce7702.ttf", "./assets/font-cbf4f5937e04.ttf", "./assets/font-d0e154fa8a11.ttf", "./assets/font-ef22c232bb39.ttf", "./assets/font-f3fb6e3aeb1b.ttf", "./assets/font-f7edf71d76fb.ttf", "./assets/font-febcd075c3e4.ttf", "./assets/fonts.css", "./assets/icon-180.png", "./assets/icon-192.png", "./assets/icon-512.png", "./assets/qrcode.js", "./bundle-info.json", "./index.html", "./manifest.json"];
self.addEventListener('install',event=>event.waitUntil(caches.open(CACHE).then(cache=>cache.addAll(FILES))));
self.addEventListener('activate',event=>event.waitUntil((async()=>{
  for(const name of await caches.keys())if(name.startsWith(PREFIX)&&name.endsWith('-'+self.registration.scope)&&name!==CACHE)await caches.delete(name);
  await self.clients.claim();
})()));
self.addEventListener('fetch',event=>{
  const url=new URL(event.request.url);
  if(event.request.method!=='GET'||url.origin!==self.location.origin||!url.href.startsWith(self.registration.scope)||url.pathname.endsWith('.zip'))return;
  event.respondWith((async()=>{
    const cache=await caches.open(CACHE);
    if(event.request.mode==='navigate'){
      try{return await fetch(event.request);}catch(error){return (await cache.match(new URL('./index.html',self.registration.scope)))||Response.error();}
    }
    const cached=await cache.match(event.request,{ignoreSearch:true});
    if(cached)return cached;
    return fetch(event.request);
  })());
});
